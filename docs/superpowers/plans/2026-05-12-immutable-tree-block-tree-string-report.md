```cpp
namespace {

using Clock = std::chrono::steady_clock;
using Entry = std::pair<std::string, std::string>;

volatile std::size_t g_size_sink = 0;

struct JemallocStats {
  std::size_t allocated = 0;
  std::size_t active = 0;
  std::size_t resident = 0;
};

struct JemallocDelta {
  std::int64_t allocated = 0;
  std::int64_t active = 0;
  std::int64_t resident = 0;
};

template <typename Tree, typename = void>
struct HasDebugStatsForTest : std::false_type {};

template <typename Tree>
struct HasDebugStatsForTest<
    Tree, std::void_t<decltype(std::declval<const Tree&>().DebugStatsForTest())>>
    : std::true_type {};

template <typename Tree>
typename std::enable_if<HasDebugStatsForTest<Tree>::value>::type AppendDebugStats(
    const Tree& tree, std::ostream& os) {
  const auto stats = tree.DebugStatsForTest();
  os << ",nodes=" << stats.node_count << ",zip_lists=" << stats.zip_list_count
     << ",entries=" << stats.entry_count << ",entry_capacity=" << stats.entry_capacity
     << ",avg_fill=" << std::fixed << std::setprecision(6) << stats.AverageFillRate()
     << std::defaultfloat << ",min_block_count=" << stats.min_block_count;
}

template <typename Tree>
typename std::enable_if<!HasDebugStatsForTest<Tree>::value>::type AppendDebugStats(
    const Tree&, std::ostream& os) {
  os << ",nodes=,zip_lists=,entries=,entry_capacity=,avg_fill=,min_block_count=";
}

std::string MakeStringKey(std::size_t index, std::size_t key_bytes) {
  const std::string prefix = "key_";
  if (key_bytes <= prefix.size()) {
    throw std::invalid_argument("key_bytes must be greater than key prefix length");
  }
  std::ostringstream os;
  os << prefix << std::setw(static_cast<int>(key_bytes - prefix.size()))
     << std::setfill('0') << index;
  return os.str();
}

std::string MakeStringValue(std::size_t index, std::size_t value_bytes) {
  const std::string prefix = "value_";
  if (value_bytes <= prefix.size() + 12) {
    throw std::invalid_argument("value_bytes must leave room for value prefix and index");
  }
  std::ostringstream os;
  os << prefix << std::setw(12) << std::setfill('0') << index;
  std::string value = os.str();
  const std::string payload = "_payload";
  while (value.size() < value_bytes) {
    value += payload;
  }
  value.resize(value_bytes);
  return value;
}

std::vector<Entry> MakeEntries(std::size_t size, const std::string& pattern,
                               std::size_t key_bytes, std::size_t value_bytes) {
  std::vector<Entry> entries;
  entries.reserve(size);
  for (std::size_t i = 0; i < size; ++i) {
    entries.push_back({MakeStringKey(i, key_bytes), MakeStringValue(i, value_bytes)});
  }
  if (pattern == "random") {
    std::mt19937 rng(0x5eed);
    std::shuffle(entries.begin(), entries.end(), rng);
  }
  return entries;
}

std::vector<std::string> MakeSortedKeys(const std::vector<Entry>& entries) {
  std::vector<std::string> keys;
  keys.reserve(entries.size());
  for (const auto& entry : entries) {
    keys.push_back(entry.first);
  }
  std::sort(keys.begin(), keys.end());
  return keys;
}

std::vector<std::string> MakeMissingKeys(std::size_t size, std::size_t key_bytes) {
  std::vector<std::string> keys;
  keys.reserve(size);
  const std::size_t offset = size + 1000000000ULL;
  for (std::size_t i = 0; i < size; ++i) {
    keys.push_back(MakeStringKey(offset + i, key_bytes));
  }
  return keys;
}

template <typename Func>
long long TimeMicros(Func func) {
  const auto start = Clock::now();
  func();
  const auto end = Clock::now();
  return std::chrono::duration_cast<std::chrono::microseconds>(end - start).count();
}

template <typename Tree>
Tree BuildTree(const std::vector<Entry>& entries) {
  Tree tree;
  for (const auto& entry : entries) {
    std::optional<Tree> next = tree.Insert(entry.first, entry.second);
    if (!next.has_value()) {
      throw std::runtime_error("duplicate insert while building key " + entry.first);
    }
    tree = *next;
  }
  if (tree.Size() != entries.size()) {
    throw std::runtime_error("tree size mismatch after build");
  }
  return tree;
}

std::size_t ReadRepetitions(std::size_t size) {
  if (size == 0) {
    return 1;
  }
  const std::size_t target_lookups = 200000;
  return std::max<std::size_t>(1, target_lookups / size);
}

#ifdef IMMUTABLE_CONTAINER_USE_JEMALLOC
void FlushJemallocThreadCache() {
  mallctl("thread.tcache.flush", nullptr, nullptr, nullptr, 0);
}

JemallocStats ReadJemallocStats() {
  std::uint64_t epoch = 1;
  std::size_t epoch_size = sizeof(epoch);
  if (mallctl("epoch", &epoch, &epoch_size, &epoch, sizeof(epoch)) != 0) {
    throw std::runtime_error("failed to refresh jemalloc epoch");
  }

  JemallocStats stats;
  std::size_t value_size = sizeof(std::size_t);
  if (mallctl("stats.allocated", &stats.allocated, &value_size, nullptr, 0) != 0) {
    throw std::runtime_error("failed to read jemalloc stats.allocated");
  }
  value_size = sizeof(std::size_t);
  if (mallctl("stats.active", &stats.active, &value_size, nullptr, 0) != 0) {
    throw std::runtime_error("failed to read jemalloc stats.active");
  }
  value_size = sizeof(std::size_t);
  if (mallctl("stats.resident", &stats.resident, &value_size, nullptr, 0) != 0) {
    throw std::runtime_error("failed to read jemalloc stats.resident");
  }
  return stats;
}
#else
void FlushJemallocThreadCache() {
  throw std::runtime_error("benchmark requires IMMUTABLE_CONTAINER_USE_JEMALLOC");
}

JemallocStats ReadJemallocStats() {
  throw std::runtime_error("benchmark requires IMMUTABLE_CONTAINER_USE_JEMALLOC");
}
#endif

JemallocDelta Delta(const JemallocStats& after, const JemallocStats& before) {
  return JemallocDelta{
      static_cast<std::int64_t>(after.allocated) - static_cast<std::int64_t>(before.allocated),
      static_cast<std::int64_t>(after.active) - static_cast<std::int64_t>(before.active),
      static_cast<std::int64_t>(after.resident) - static_cast<std::int64_t>(before.resident)};
}

template <typename Tree>
void RunCase(const std::string& name, const std::string& pattern, std::size_t size,
             std::size_t key_bytes, std::size_t value_bytes) {
  const std::vector<Entry> entries = MakeEntries(size, pattern, key_bytes, value_bytes);
  const std::vector<std::string> sorted_keys = MakeSortedKeys(entries);
  const std::vector<std::string> missing_keys = MakeMissingKeys(size, key_bytes);

  FlushJemallocThreadCache();
  const JemallocStats before = ReadJemallocStats();

  Tree tree;
  const long long build_us = TimeMicros([&] { tree = BuildTree<Tree>(entries); });

  FlushJemallocThreadCache();
  const JemallocStats after_build = ReadJemallocStats();
  const JemallocDelta memory_delta = Delta(after_build, before);

  const std::size_t repetitions = ReadRepetitions(size);

  const long long hit_contains_us = TimeMicros([&] {
    std::size_t hits = 0;
    for (std::size_t rep = 0; rep < repetitions; ++rep) {
      for (const auto& key : sorted_keys) {
        if (tree.Contains(key)) {
          ++hits;
        }
      }
    }
    g_size_sink += hits;
  });

  const long long miss_contains_us = TimeMicros([&] {
    std::size_t misses = 0;
    for (std::size_t rep = 0; rep < repetitions; ++rep) {
      for (const auto& key : missing_keys) {
        if (!tree.Contains(key)) {
          ++misses;
        }
      }
    }
    g_size_sink += misses;
  });

  const long long to_vector_us = TimeMicros([&] {
    const auto values = tree.ToVector();
    g_size_sink += values.size();
  });

  std::cout << "case,name=" << name << ",pattern=" << pattern << ",size=" << size
            << ",key_bytes=" << key_bytes << ",value_bytes=" << value_bytes
            << ",repetitions=" << repetitions << ",height=" << tree.Height()
            << ",build_us=" << build_us << ",hit_contains_us=" << hit_contains_us
            << ",miss_contains_us=" << miss_contains_us << ",to_vector_us=" << to_vector_us
            << ",allocated_start=" << before.allocated
            << ",allocated_after_build=" << after_build.allocated
            << ",allocated_delta=" << memory_delta.allocated
            << ",active_start=" << before.active
            << ",active_after_build=" << after_build.active
            << ",active_delta=" << memory_delta.active
            << ",resident_start=" << before.resident
            << ",resident_after_build=" << after_build.resident
            << ",resident_delta=" << memory_delta.resident;
  AppendDebugStats(tree, std::cout);
  std::cout << "\n";
}

template <typename Tree>
void RunTree(const std::string& name) {
  for (const char* pattern : {"sorted", "random"}) {
    for (std::size_t key_bytes : {32, 64}) {
      for (std::size_t value_bytes : {64, 128, 256, 1024}) {
        for (std::size_t size : {1, 10, 100, 1000, 10000, 100000}) {
          RunCase<Tree>(name, pattern, size, key_bytes, value_bytes);
        }
      }
    }
  }
}
```

```python
#!/usr/bin/env python3
import argparse
import datetime as _datetime
import html
from pathlib import Path
import sys


PERF_FIELDS = [
    ("build_us", "Build"),
    ("hit_contains_us", "Contains hit"),
    ("miss_contains_us", "Contains miss"),
    ("to_vector_us", "ToVector"),
]

MEMORY_FIELDS = [
    ("allocated_delta", "Allocated delta"),
    ("active_delta", "Active delta"),
    ("resident_delta", "Resident delta"),
]


def parse_line(line):
    parts = line.strip().split(",")
    if not parts or not parts[0]:
        return None
    row = {"row_type": parts[0]}
    for part in parts[1:]:
        if "=" not in part:
            raise ValueError(f"malformed field {part!r}")
        key, value = part.split("=", 1)
        row[key] = value
    return row


def read_rows(path):
    env_rows = []
    case_rows = []
    for line_number, line in enumerate(Path(path).read_text(encoding="utf-8").splitlines(), 1):
        if not line.strip():
            continue
        row = parse_line(line)
        if row["row_type"] == "env":
            env_rows.append(row)
        elif row["row_type"] == "case":
            case_rows.append(row)
        else:
            raise ValueError(f"{path}:{line_number}: unknown row type {row['row_type']!r}")
    if not case_rows:
        raise ValueError(f"{path}: no case rows found")
    required = {
        "name",
        "pattern",
        "size",
        "key_bytes",
        "value_bytes",
        "height",
        "build_us",
        "hit_contains_us",
        "miss_contains_us",
        "to_vector_us",
        "allocated_delta",
        "active_delta",
        "resident_delta",
    }
    for index, row in enumerate(case_rows, 1):
        missing = sorted(required - set(row))
        if missing:
            raise ValueError(f"case row {index} missing fields: {', '.join(missing)}")
    return env_rows, case_rows


def as_int(row, key):
    value = row.get(key, "")
    if value == "":
        return None
    return int(value)


def as_float(row, key):
    value = row.get(key, "")
    if value == "":
        return None
    return float(value)


def fmt_int(value):
    if value is None:
        return ""
    return f"{value:,}"


def fmt_us(value):
    if value is None:
        return ""
    return f"{value:,} us"


def fmt_bytes(value):
    if value is None:
        return ""
    sign = "-" if value < 0 else ""
    abs_value = abs(value)
    units = ["B", "KiB", "MiB", "GiB"]
    amount = float(abs_value)
    unit = units[0]
    for unit in units:
        if amount < 1024.0 or unit == units[-1]:
            break
        amount /= 1024.0
    if unit == "B":
        return f"{sign}{int(amount)} {unit}"
    return f"{sign}{amount:.2f} {unit}"


def fmt_bytes_per_entry(value, size):
    if value is None or size <= 0:
        return ""
    return f"{value / size:.1f} B/entry"


def fmt_float(value):
    if value is None:
        return ""
    return f"{value:.3f}"


def sort_cases(cases):
    order = {"immutable_tree": 0, "block_tree_2048": 1, "block_tree_4096": 2}
    return sorted(
        cases,
        key=lambda row: (
            row["pattern"],
            as_int(row, "key_bytes"),
            as_int(row, "value_bytes"),
            as_int(row, "size"),
            order.get(row["name"], 99),
        ),
    )


def group_patterns(cases):
    patterns = []
    for row in sort_cases(cases):
        if row["pattern"] not in patterns:
            patterns.append(row["pattern"])
    return patterns


def ratio(candidate, baseline):
    if candidate is None or baseline in (None, 0):
        return ""
    return f"{candidate / baseline:.2f}x"


def build_index(cases):
    result = {}
    for row in cases:
        result[
            (
                row["pattern"],
                as_int(row, "key_bytes"),
                as_int(row, "value_bytes"),
                as_int(row, "size"),
                row["name"],
            )
        ] = row
    return result


def make_observations(cases):
    index = build_index(cases)
    observations = []
    max_sizes = sorted({as_int(row, "size") for row in cases})
    if not max_sizes:
        return observations
    size = max_sizes[-1]
    for pattern in group_patterns(cases):
        key_lengths = sorted({as_int(row, "key_bytes") for row in cases if row["pattern"] == pattern})
        value_lengths = sorted({as_int(row, "value_bytes") for row in cases if row["pattern"] == pattern})
        for key_bytes in key_lengths:
            for value_bytes in value_lengths:
                baseline = index.get((pattern, key_bytes, value_bytes, size, "immutable_tree"))
                if baseline is None:
                    continue
                base_build = as_int(baseline, "build_us")
                base_alloc = as_int(baseline, "allocated_delta")
                for name in ("block_tree_2048", "block_tree_4096"):
                    candidate = index.get((pattern, key_bytes, value_bytes, size, name))
                    if candidate is None:
                        continue
                    build_text = ratio(as_int(candidate, "build_us"), base_build)
                    alloc_text = ratio(as_int(candidate, "allocated_delta"), base_alloc)
                    observations.append(
                        f"For {pattern} insertion at {size:,} entries with {key_bytes}-byte keys "
                        f"and {value_bytes}-byte values, {name} build time is "
                        f"{build_text or 'n/a'} of immutable_tree and allocated memory is "
                        f"{alloc_text or 'n/a'} of immutable_tree."
                    )
    return observations


def table(headers, rows):
    out = ["<table>", "<thead><tr>"]
    out.extend(f"<th>{html.escape(header)}</th>" for header in headers)
    out.append("</tr></thead>")
    out.append("<tbody>")
    for row in rows:
        out.append("<tr>")
        out.extend(f"<td>{html.escape(str(cell))}</td>" for cell in row)
        out.append("</tr>")
    out.append("</tbody></table>")
    return "\n".join(out)


def performance_table(cases, pattern):
    rows = []
    for row in sort_cases([case for case in cases if case["pattern"] == pattern]):
        size = as_int(row, "size")
        rows.append([
            row["name"],
            fmt_int(as_int(row, "key_bytes")),
            fmt_int(as_int(row, "value_bytes")),
            fmt_int(size),
            fmt_int(as_int(row, "height")),
            fmt_int(as_int(row, "repetitions")),
            *[fmt_us(as_int(row, field)) for field, _ in PERF_FIELDS],
        ])
    return table(
        ["Implementation", "Key bytes", "Value bytes", "Size", "Height", "Lookup reps"]
        + [label for _, label in PERF_FIELDS],
        rows,
    )


def memory_table(cases, pattern):
    rows = []
    for row in sort_cases([case for case in cases if case["pattern"] == pattern]):
        size = as_int(row, "size")
        cells = [
            row["name"],
            fmt_int(as_int(row, "key_bytes")),
            fmt_int(as_int(row, "value_bytes")),
            fmt_int(size),
        ]
        for field, _ in MEMORY_FIELDS:
            value = as_int(row, field)
            cells.append(fmt_bytes(value))
            cells.append(fmt_bytes_per_entry(value, size))
        rows.append(cells)
    headers = ["Implementation", "Key bytes", "Value bytes", "Size"]
    for _, label in MEMORY_FIELDS:
        headers.extend([label, f"{label} per entry"])
    return table(headers, rows)


def structure_table(cases, pattern):
    rows = []
    for row in sort_cases([case for case in cases if case["pattern"] == pattern]):
        rows.append([
            row["name"],
            fmt_int(as_int(row, "key_bytes")),
            fmt_int(as_int(row, "value_bytes")),
            fmt_int(as_int(row, "size")),
            fmt_int(as_int(row, "nodes")),
            fmt_int(as_int(row, "zip_lists")),
            fmt_int(as_int(row, "entries")),
            fmt_int(as_int(row, "entry_capacity")),
            fmt_float(as_float(row, "avg_fill")),
            fmt_int(as_int(row, "min_block_count")),
        ])
    return table(
        [
            "Implementation",
            "Key bytes",
            "Value bytes",
            "Size",
            "Nodes",
            "Zip lists",
            "Entries",
            "Entry capacity",
            "Average fill",
            "Minimum block count",
        ],
        rows,
    )


def render_report(env_rows, cases, args):
    generated_at = _datetime.datetime.now(_datetime.timezone.utc).astimezone().isoformat(timespec="seconds")
    observations = make_observations(cases)
    env = env_rows[0] if env_rows else {}
    patterns = group_patterns(cases)
    sections = []
    for pattern in patterns:
        sections.append(f"<h2>{html.escape(pattern.title())} Inserts</h2>")
        sections.append("<h3>Performance</h3>")
        sections.append(performance_table(cases, pattern))
        sections.append("<h3>jemalloc Memory</h3>")
        sections.append(memory_table(cases, pattern))
        sections.append("<h3>Block Structure</h3>")
        sections.append(structure_table(cases, pattern))

    observation_items = "".join(f"<li>{html.escape(item)}</li>" for item in observations)
    if not observation_items:
        observation_items = "<li>No baseline observations could be computed from the input rows.</li>"

    metadata_rows = [
        ["Generated at", generated_at],
        ["Input", str(args.input)],
        ["Benchmark command", args.bench_command],
        ["jemalloc path", args.jemalloc_path],
        ["Allocator", env.get("allocator", "jemalloc")],
        ["Key type", env.get("key_type", "std::string")],
        ["Value type", env.get("value_type", "std::string")],
        ["Key byte lengths", env.get("key_bytes", "32|64")],
        ["Value byte lengths", env.get("value_bytes", "64|128|256|1024")],
        ["Sizes", env.get("sizes", "1|10|100|1000|10000|100000")],
        ["Patterns", env.get("patterns", "sorted|random")],
    ]

    return f"""<!doctype html>
<html lang="en">
<head>
  <meta charset="utf-8">
  <title>ImmutableTree vs ImmutableBlockTree String Report</title>
  <style>
    body {{
```
